\cventry
{هوش مصنوعی سلامت آمازون \lr{|} بنچمارک‌های ارزیابی عامل‌های هوش مصنوعی سلامت}
{\lr{PatientAgentBench} و \lr{IMCBench}: ارزیابی عامل‌های هوش مصنوعی سلامت در تعامل با بیمار و مبتنی بر تصویر}
{هوش مصنوعی سلامت آمازون}
{۲۰۲۵ \lr{--} ۲۰۲۶}
{
\begin{cvitems}
\item {رهبری دو مطالعه بنچمارک برای ارزیابی عامل‌های هوش مصنوعی سلامت در تعامل با بیمار و در گفتگوهای چندوجهی: \href{https://arxiv.org/abs/2607.25485}{\lr{PatientAgentBench}} (\lr{arXiv:2607.25485}، ژوئیه ۲۰۲۶)، ارزیابی ۱۰ مدل روی ۱٬۲۰۰ سناریوی شبیه‌سازی‌شده بیمار، و \href{https://arxiv.org/abs/2606.28556}{\lr{IMCBench}} (\lr{arXiv:2606.28556}، ژوئن ۲۰۲۶)، ارزیابی ۸ مدل چندوجهی در گفتگوهای پزشکی مبتنی بر تصویر با ترکیب تصاویر بالینی، پروفایل بیمار و زمینه \lr{EHR}.}
\item {هر دو چارچوب با روش \textbf{\lr{LLM-as-a-Jury}} اعتبارسنجی‌شده در برابر پزشکان دارای مجوز (توافق ۷۹ تا ۹۳ درصد) در ابعادی چون ایمنی، اتکاپذیری به منابع، عدم‌قطعیت تشخیصی و استدلال بالینی چندنوبتی عمل می‌کنند.}
\item {یافته‌ها نشان می‌دهند عملکرد قوی در وظایف محدود (مانند توصیف تصویر) لزوماً به معنای راهنمایی ایمن برای بیمار نیست و بنچمارک‌های ایستا و تک‌نوبتی برای ارزیابی نسل جدید عامل‌های سلامت کافی نیستند \lr{--} یافته‌ای که برنامه گسترده‌تر ارزیابی و بنچمارک‌گذاری هوش مصنوعی سلامت آمازون را هدایت کرده است.}
\end{cvitems}
}
