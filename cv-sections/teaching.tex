\cvsection{تدریس، طراحی برنامه درسی و رهبری آموزشی}

\begin{cventries}

\cventry
{دستیار آموزشی تحصیلات تکمیلی — علوم کامپیوتر و مهندسی}
{دانشگاه بوفالو، دانشگاه ایالتی نیویورک}
{بوفالو، نیویورک، آمریکا}
{ژانویه ۲۰۱۸ – مه ۲۰۲۱}
{
\begin{cvitems}
\item {تدریس و پشتیبانی آموزشی در دوره‌های علوم کامپیوتر و هوش مصنوعی در مقطع تحصیلات تکمیلی و کارشناسی.}
\item {راهنمایی دانشجویان در پروژه‌های یادگیری ماشین و یادگیری عمیق، کمک به درک مفاهیم نظری و پیاده‌سازی مدل‌ها.}
\item {ارزیابی پروژه‌ها و تکالیف و ارائه بازخورد فنی برای تقویت مهارت‌های حل مسئله و پژوهش دانشجویان.}
\item {تجربه مستقیم در تبدیل مفاهیم پیچیده هوش مصنوعی به محتوای آموزشی و فعالیت‌های عملی قابل استفاده برای دانشجویان با سطوح مختلف دانش فنی.}
\end{cvitems}
}

\cventry
{طراحی کارگاه و برنامه آموزشی یادگیری عمیق}
{انجمن دانشجویان تحصیلات تکمیلی علوم کامپیوتر و مهندسی \lr{(CSEGSA)} — دانشگاه بوفالو}
{بوفالو، نیویورک، آمریکا}
{۲۰۱۹ – ۲۰۲۰}
{
\begin{cvitems}
\item {طراحی و سازمان‌دهی کارگاه‌های جامع «یادگیری عمیق کاربردی» برای آشنایی دانشجویان با مبانی، معماری‌ها، ابزارها و کاربردهای یادگیری عمیق.}
\item {ایجاد بستری برای انتقال دانش میان پژوهشگران دانشگاهی و متخصصان صنعت و پیوند دادن محتوای آموزشی با مسائل واقعی پژوهش و صنعت.}
\item {رهبری انجمن به‌عنوان رئیس \lr{CSEGSA} و برنامه‌ریزی رویدادهای آموزشی و اجتماعی برای جامعه دانشجویان تحصیلات تکمیلی علوم کامپیوتر.}
\end{cvitems}
}

\cventry
{رویکرد آموزشی و برنامه درسی پیشنهادی برای هوش مصنوعی نسل جدید}
{تمرکز تخصصی}
{هوش مصنوعی، \lr{Agentic AI} و پژوهش کاربردی}
{اکنون}
{
\begin{cvitems}
\item {تمرکز آموزشی بر پیوند میان مبانی هوش مصنوعی و فناوری‌های نسل جدید شامل \lr{LLM}، \lr{RAG}، \lr{Agentic AI}، ارزیابی مدل، ایمنی و کاربردهای واقعی.}
\item {تجربه صنعت در توسعه سامانه‌های هوشمند در مقیاس تولید برای تبدیل مسائل واقعی به پروژه‌ها و نمونه‌های آموزشی مبتنی بر مسئله.}
\item {رویکرد برنامه‌درسی مبتنی بر ترکیب \textbf{مبانی نظری + پژوهش + پروژه عملی + ارزیابی تجربی + اخلاق و ایمنی هوش مصنوعی}.}
\end{cvitems}
}

\end{cventries}
