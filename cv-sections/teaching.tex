{ژانویه ۲۰۱۸ – مه ۲۰۲۱}
{
\begin{cvitems}
\item {تدریس درس مبانی یادگیری ماشین \lr{(CSE 574)} برای کلاسی با بیش از ۱۰۰ دانشجو و ارائه مفاهیم پایه یادگیری ماشین.}
\item {تدریس درس یادگیری عمیق \lr{(CSE 676)} در چهار نیم‌سال (پاییز ۲۰۱۸، پاییز ۲۰۱۹، بهار ۲۰۲۰ و پاییز ۲۰۲۰)، شامل شبکه‌های عصبی بازگشتی و مدل‌های مبتنی بر توجه، برای کلاس‌هایی با ۹۰ تا بیش از ۱۰۰ دانشجو، و هماهنگی بیش از ۵۰ گروه پروژه کارشناسی ارشد.}
\item {تدریس درس یادگیری ماشین پیشرفته \lr{(CSE 674)} با تمرکز بر شبکه‌های مولد تخاصمی \lr{(GAN)} برای کلاسی با بیش از ۵۰ دانشجو.}
\item {راهنمایی دانشجویان در پروژه‌های یادگیری ماشین و یادگیری عمیق، ارزیابی پروژه‌ها و تکالیف، و تبدیل مفاهیم پیچیده هوش مصنوعی به محتوای آموزشی و فعالیت‌های عملی کاربردی.}
\end{cvitems}
}

{
\begin{cvitems}

\item {ایجاد بستری برای انتقال دانش میان پژوهشگران دانشگاهی و متخصصان صنعت و پیوند دادن محتوای آموزشی با مسائل واقعی پژوهش و صنعت.}
\item {رهبری انجمن به‌عنوان رئیس \lr{CSEGSA} و برنامه‌ریزی رویدادهای آموزشی و اجتماعی برای جامعه دانشجویان تحصیلات تکمیلی علوم کامپیوتر.}
\end{cvitems}
}
