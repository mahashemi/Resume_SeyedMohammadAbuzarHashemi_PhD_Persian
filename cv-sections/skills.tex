\cvsection{مهارت‌های هوش مصنوعی، پژوهش، آموزش و رهبری}

\begin{cvskills}
\cvskill{هوش مصنوعی عامل‌محور}{هوش مصنوعی عامل‌محور، عامل‌های \lr{LLM}، استفاده از ابزار، هماهنگ‌سازی عامل‌ها، \lr{RAG}، مهندسی درخواست، ارزیابی و حفاظ‌های ایمنی}
\cvskill{پژوهش و مدل‌سازی}{\lr{LLM}، یادگیری چندوجهی، بینایی رایانه‌ای، \lr{NLP}، یادگیری خودنظارتی، یادگیری انتقالی، تنظیم دقیق و \lr{LoRA}}
\cvskill{مهندسی هوش مصنوعی}{پایتون، پای‌تورچ، تنسورفلو، کراس، بردارهای تعبیه‌شده، جست‌وجوی برداری، \lr{FAISS}، ارزیابی مدل و \lr{MLOps}}
\cvskill{پژوهش و نوآوری}{طراحی آزمایش، ارزیابی مدل، پژوهش کاربردی، انتشار علمی، ثبت اختراع و تبدیل پژوهش به محصول}
\cvskill{آموزش و برنامه درسی}{تدریس هوش مصنوعی، طراحی کارگاه، یادگیری پروژه‌محور، طراحی محتوای آموزشی و طراحی برنامه درسی}
\cvskill{رهبری و مدیریت فنی}{تعریف مسئله، جهت‌دهی پژوهش، رهبری پروژه، همکاری میان‌رشته‌ای، راهنمایی دانشجویان و اعضای تیم، ساخت تیم و انتقال دانش}
\end{cvskills}
