\begin{cvitems}
\item {هدایت انجمن دانشجویان تحصیلات تکمیلی علوم کامپیوتر و مهندسی و ایجاد برنامه‌هایی برای پیوند دانشجویان با پژوهشگران دانشگاهی و متخصصان صنعت.}
\item {طراحی و سازمان‌دهی برنامه‌های آموزشی، کارگاه‌ها، رویدادهای معارفه و نشست‌های شبکه‌سازی برای تقویت مشارکت و توسعه حرفه‌ای جامعه دانشجویان تحصیلات تکمیلی.}
\item {ایجاد فرصت‌هایی برای تعامل میان دانشجویان، اعضای هیئت علمی، پژوهشگران و متخصصان صنعت و تبدیل نیازهای جامعه دانشجویی به برنامه‌های آموزشی و اجتماعی.}
\item {دریافت جایزه رهبری دانشجویان تحصیلات تکمیلی علوم کامپیوتر و مهندسی دانشگاه بوفالو در سال ۲۰۲۰ در تقدیر از مشارکت و خدمت به جامعه دانشجویی.}
\end{cvitems}
}
