\cventry
{دانشمند کاربردی \lr{·} هوش مصنوعی سلامت \lr{·} پژوهش و هوش مصنوعی عامل‌محور}
{هوش مصنوعی سلامت آمازون}
{سیاتل، واشینگتن، آمریکا}
{ژوئن ۲۰۲۱ \lr{—} اکنون}
{
\begin{cvitems}
\item {مشارکت علمی کلیدی در توسعه و راه‌اندازی \lr{Amazon Health AI} و دستیار سلامت مبتنی بر هوش مصنوعی عامل‌محور \lr{Amazon One Medical} در سال ۲۰۲۶، و گسترش آن به \lr{Amazon.com} و اپلیکیشن آمازون برای مشتریان آمازون و اعضای واجد شرایط \lr{Prime}.}
\item {تعریف مسائل پژوهشی، تعیین جهت‌گیری فنی، طراحی آزمایش‌ها، انتخاب مدل‌ها و معماری‌ها، و هدایت پژوهش از نمونه‌سازی تا ارزیابی کمی و استقرار در محیط تولید.}
\item {هدایت پژوهش و ارزیابی سامانه‌های هوش مصنوعی سلامت در مقیاس تولید، با تمرکز بر قابلیت اعتماد، اتکاپذیری به منابع، ایمنی، مقاومت در برابر تزریق دستور، تشخیص‌های قطعی ناخواسته، تعاملات چندمرحله‌ای، زمینه بالینی و استفاده از ابزارها.}
\item {مشارکت در پژوهش‌های داوری‌شده درباره بنچمارک سامانه‌های هوش مصنوعی سلامت بیمارمحور و چندوجهی، از جمله \lr{PatientAgentBench} و \lr{IMCBench}.}
\item {همکاری میان تیم‌های پژوهش، مهندسی، محصول و سلامت برای تبدیل پژوهش‌های نوین هوش مصنوعی به قابلیت‌های قابل اتکا و هدایت مسائل اثرگذار از پژوهش تا استقرار.}
\item {طراحی و توسعه سامانه‌های مبتنی بر \lr{LLM}، \lr{RAG} و هوش مصنوعی عامل‌محور برای مسائل پیچیده سلامت، شامل بازنویسی پرس‌وجو، بازیابی، هماهنگی ابزارها، ارزیابی پاسخ و حفاظ‌های ایمنی.}
\end{cvitems}
}

\cventry
{پژوهش دکتری \lr{·} ارنست اند یانگ / آزمایشگاه‌های نوآوری آر‌بی‌سی}
{پژوهش و نوآوری هوش مصنوعی صنعتی}
{پالو آلتو و اورلاندو، آمریکا}
{مه ۲۰۱۸ \lr{—} اوت ۲۰۱۹}
{
\begin{cvitems}
\item {هم‌مخترع چند خانواده ثبت اختراع در ایالات متحده که از پژوهش‌های هوش مصنوعی در آزمایشگاه‌های نوآوری آر‌بی‌سی حاصل شدند، در حوزه‌هایی شامل اعتبارنامه‌های توکن‌شده امن، احراز هویت زیستی و گذرواژه‌های پویا.}
\item {پژوهش و پیاده‌سازی مدل‌های ترنسفورمر چندوجهی، شبکه‌های مولد تخاصمی \lr{(GAN)}، شبکه‌های \lr{Conv-LSTM} و مدل‌های بینایی رایانه‌ای و پردازش زبان طبیعی برای مسائل هوشمندی اسناد، تشخیص گفتار از روی حرکت لب و احراز هویت چهره.}
\item {توسعه بردارهای تعبیه‌شده برای اشیای مشابه در اسناد تجاری اسکن‌شده و دستیابی به دقت طبقه‌بندی ۹۲٪ و نتیجه جدید \lr{SOTA} در مجموعه‌داده اسناد.}
\item {توسعه سامانه استخراج موجودیت از رسیدهای اسکن‌شده و سامانه احراز هویت چهره برای بانکداری امن و بدون تماس؛ سامانه احراز هویت به‌صورت داخلی مستقر شد.}
\end{cvitems}
}
