\cvsection{سوابق حرفه‌ای و پژوهشی}

\begin{cventries}

\cventry
{دانشمند کاربردی دوم \lr{—} هوش مصنوعی و سلامت \lr{|} پژوهش، هوش مصنوعی عامل‌محور و سامانه‌های هوشمند \lr{—} تمام‌وقت}
{آمازون}
{سیاتل، واشینگتن، آمریکا}
{ژوئن ۲۰۲۱ \lr{—} اکنون}
{
\begin{cvitems}
\item {مشارکت علمی کلیدی در توسعه و عرضه دستیار سلامت عامل‌محور در آمازون وان مدیکال در سال ۲۰۲۶ و سپس گسترش هوش مصنوعی سلامت به آمازون و اپلیکیشن آمازون برای ارائه راهنمایی شخصی‌سازی‌شده سلامت و دسترسی ۲۴\lr{/}۷ به خدمات سلامت.}
\item {طراحی و توسعه راهکارهای مبتنی بر \textbf{\lr{LLM}، \lr{RAG} و هوش مصنوعی عامل‌محور} برای مسائل پیچیده سلامت، شامل بازنویسی درخواست، بازیابی دانش، هماهنگ‌سازی ابزارها، ارزیابی پاسخ و حفاظ‌های ایمنی.}
\item {پژوهش و ارزیابی سامانه‌های هوشمند در مقیاس تولید با تمرکز بر قابلیت اتکا، کیفیت پاسخ، مقاومت در برابر تزریق درخواست، ایمنی و کاهش پاسخ‌های تشخیصی قطعی ناخواسته.}
\item {مشارکت در پژوهش و توسعه چارچوب‌های ارزیابی نسل جدید سامانه‌های هوش مصنوعی سلامت، از جمله ارزیابی ایمنی، تعامل چندمرحله‌ای، استفاده از ابزارها، زمینه بالینی و عملکرد عامل‌ها در سناریوهای واقعی بیماران.}
\item {همکاری با پژوهشگران، مهندسان و متخصصان حوزه سلامت برای تبدیل ایده‌های پژوهشی و نمونه‌های اولیه به قابلیت‌های قابل اتکا در محصول و هدایت مسائل از پژوهش تا استقرار تولیدی.}
\item {تعریف مسائل، طراحی آزمایش، انتخاب مدل و معماری، توسعه نمونه اولیه، ارزیابی کمی و هدایت تصمیم‌های فنی در مسیر تولید.}
\end{cvitems}
}

\cventry
{پژوهش صنعتی هوش مصنوعی در دوره دکتری \lr{—} بینایی رایانه‌ای، پردازش زبان طبیعی و هوشمندی اسناد}
{ارنست اند یانگ \lr{·} آزمایشگاه‌های نوآوری آر‌بی‌سی}
{پالو آلتو و اورلاندو، آمریکا}
{مه ۲۰۱۸ \lr{—} اوت ۲۰۱۹}
{
\begin{cvitems}
\item {پژوهش و پیاده‌سازی مدل‌های ترنسفورمر چندوجهی، شبکه‌های مولد تخاصمی \lr{(GAN)}، شبکه‌های \lr{Conv-LSTM} و مدل‌های بینایی رایانه‌ای و پردازش زبان طبیعی برای مسائل هوشمندی اسناد، تشخیص گفتار از روی حرکت لب و احراز هویت چهره.}
\item {توسعه بردارهای تعبیه‌شده برای اشیای مشابه در اسناد تجاری اسکن‌شده و دستیابی به دقت طبقه‌بندی ۹۲٪ و نتیجه جدید \lr{SOTA} در مجموعه‌داده اسناد.}
\item {توسعه سامانه استخراج موجودیت از رسیدهای اسکن‌شده و سامانه احراز هویت چهره برای بانکداری امن و بدون تماس؛ سامانه احراز هویت به‌صورت داخلی مستقر شد.}
\end{cvitems}
}
