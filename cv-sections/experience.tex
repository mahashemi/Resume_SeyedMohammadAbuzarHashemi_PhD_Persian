\cvsection{سوابق حرفه‌ای و پژوهشی}

\begin{cventries}

\cventry
{دانشمند کاربردی دوم \lr{—} هوش مصنوعی و سلامت \lr{|} پژوهش، هوش مصنوعی عامل‌محور و سامانه‌های هوشمند \lr{—} تمام‌وقت}
{آمازون}
{سیاتل، واشینگتن، آمریکا}
{ژوئن ۲۰۲۱ \lr{—} اکنون}
{
\begin{cvitems}
\item {مشارکت علمی کلیدی در توسعه و عرضه هوش مصنوعی سلامت آمازون و دستیار سلامت عامل‌محور آمازون وان مدیکال در سال ۲۰۲۶ و سپس گسترش آن به آمازون و اپلیکیشن آمازون، برای ارائه راهنمایی شخصی‌سازی‌شده سلامت و دسترسی شبانه‌روزی به خدمات سلامت.}
\item {تعریف مسائل پژوهشی، طراحی آزمایش‌ها، انتخاب مدل و معماری، توسعه نمونه‌های اولیه، ارزیابی کمی و هدایت تصمیم‌های فنی از پژوهش تا قابلیت‌های تولیدی.}
\item {پژوهش و ارزیابی سامانه‌های هوش مصنوعی سلامت در مقیاس تولید، با تمرکز بر قابلیت اتکا، زمینه‌مندی، ایمنی، مقاومت در برابر تزریق درخواست، تعامل چندمرحله‌ای، استفاده از ابزارها و کاهش رفتارهای ناخواسته از جمله پاسخ‌های تشخیصی قطعی.}
\item {مشارکت در پژوهش و توسعه چارچوب‌های ارزیابی نسل جدید عامل‌های هوش مصنوعی سلامت، از جمله پژوهش‌های منتشرشده درباره ارزیابی عامل‌های سلامت بیمارمحور و چندوجهی.}
\item {همکاری میان پژوهش، مهندسی و حوزه سلامت برای تبدیل ایده‌های پژوهشی به قابلیت‌های قابل اتکا در محصول و هدایت مسائل از پژوهش تا استقرار.}
\item {طراحی و توسعه راهکارهای مبتنی بر \textbf{\lr{LLM}، \lr{RAG} و هوش مصنوعی عامل‌محور} برای مسائل پیچیده سلامت، شامل بازنویسی درخواست، بازیابی دانش، هماهنگ‌سازی ابزارها، ارزیابی پاسخ و حفاظ‌های ایمنی.}
\end{cvitems}
}

\cventry
{پژوهش صنعتی هوش مصنوعی در دوره دکتری \lr{—} بینایی رایانه‌ای، پردازش زبان طبیعی و هوشمندی اسناد}
{ارنست اند یانگ \lr{·} آزمایشگاه‌های نوآوری آر‌بی‌سی}
{پالو آلتو و اورلاندو، آمریکا}
{مه ۲۰۱۸ \lr{—} اوت ۲۰۱۹}
{
\begin{cvitems}
\item {هم‌مخترع چند خانواده ثبت اختراع در ایالات متحده که از پژوهش‌های هوش مصنوعی در آزمایشگاه‌های نوآوری آر‌بی‌سی حاصل شدند، در حوزه‌هایی شامل اعتبارنامه‌های توکن‌شده امن، احراز هویت زیستی و گذرواژه‌های پویا.}
\item {پژوهش و پیاده‌سازی مدل‌های ترنسفورمر چندوجهی، شبکه‌های مولد تخاصمی \lr{(GAN)}، شبکه‌های \lr{Conv-LSTM} و مدل‌های بینایی رایانه‌ای و پردازش زبان طبیعی برای مسائل هوشمندی اسناد، تشخیص گفتار از روی حرکت لب و احراز هویت چهره.}
\item {توسعه بردارهای تعبیه‌شده برای اشیای مشابه در اسناد تجاری اسکن‌شده و دستیابی به دقت طبقه‌بندی ۹۲٪ و نتیجه جدید \lr{SOTA} در مجموعه‌داده اسناد.}
\item {توسعه سامانه استخراج موجودیت از رسیدهای اسکن‌شده و سامانه احراز هویت چهره برای بانکداری امن و بدون تماس؛ سامانه احراز هویت به‌صورت داخلی مستقر شد.}
\end{cvitems}
}
